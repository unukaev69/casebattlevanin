function initAudio() {
if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
if (audioCtx.state === 'suspended') audioCtx.resume();
}
function playCoinSound() {
initAudio();
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = 'sine';
osc.frequency.setValueAtTime(980, audioCtx.currentTime);
osc.frequency.setValueAtTime(1310, audioCtx.currentTime + 0.04);
gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.09);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + 0.09);
}
function playTick() {
initAudio();
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = 'triangle';
osc.frequency.setValueAtTime(600, audioCtx.currentTime);
gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.04);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + 0.04);
}
function playWinSound(tier) {
initAudio();
const osc = audioCtx.createOscillator();
const gain = audioCtx.createGain();
osc.type = tier === 'ultra' ? 'sawtooth' : 'sine';
if (tier === 'ultra') {
osc.frequency.setValueAtTime(523, audioCtx.currentTime);
osc.frequency.setValueAtTime(659, audioCtx.currentTime + 0.1);
osc.frequency.setValueAtTime(784, audioCtx.currentTime + 0.2);
osc.frequency.setValueAtTime(1046, audioCtx.currentTime + 0.3);
} else if (tier === 'profit') {
osc.frequency.setValueAtTime(440, audioCtx.currentTime);
osc.frequency.setValueAtTime(659, audioCtx.currentTime + 0.15);
} else {
osc.frequency.setValueAtTime(300, audioCtx.currentTime);
osc.frequency.setValueAtTime(240, audioCtx.currentTime + 0.15);
}
gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.55);
osc.connect(gain);
gain.connect(audioCtx.destination);
osc.start();
osc.stop(audioCtx.currentTime + 0.55);
}

// Генерация предмета строго из категории выбранного кейса
function generateDropForCase(caseObj) {
const roll = Math.random() * 100;
const basePrice = caseObj.price;
const catPool = poolsByCategory[caseObj.category] || poolsByCategory.weapons;
if (roll < 0.1) {
const pool = [...catPool.ultra, caseObj.special];
const template = pool[Math.floor(Math.random() * pool.length)];
const multiplier = 12 + Math.random() * 18;
return {
name: template.name,
emoji: template.emoji,
rarity: "gold",
price: Math.round(basePrice * multiplier),
tier: "ultra"
};
} else if (roll < 40.0) {
const template = catPool.profit[Math.floor(Math.random() * catPool.profit.length)];
const multiplier = 1.2 + Math.random() * 2.3;
return {
name: template.name,
emoji: template.emoji,
rarity: template.rarity,
price: Math.max(basePrice + 2, Math.round(basePrice * multiplier)),
tier: "profit"
};
} else {
const template = catPool.cheap[Math.floor(Math.random() * catPool.cheap.length)];
const multiplier = 0.2 + Math.random() * 0.6;
return {
name: template.name,
emoji: template.emoji,
rarity: template.rarity,
price: Math.max(1, Math.min(basePrice - 1, Math.round(basePrice * multiplier))),
tier: "cheap"
};
}
}

function generateSkinByTargetPrice(targetPrice) {
const wPool = poolsByCategory.weapons;
let pool = wPool.cheap;
let rarity = "blue";
if (targetPrice >= 2500) {
pool = wPool.ultra;
rarity = "gold";
} else if (targetPrice >= 500) {
pool = wPool.profit.filter(s => s.rarity === 'red');
rarity = "red";
} else if (targetPrice >= 120) {
pool = wPool.profit.filter(s => s.rarity === 'pink');
rarity = "pink";
} else if (targetPrice >= 40) {
pool = wPool.cheap.filter(s => s.rarity === 'purple');
rarity = "purple";
}
if (pool.length === 0) pool = wPool.profit;
const chosen = pool[Math.floor(Math.random() * pool.length)];
return {
name: chosen.name,
emoji: chosen.emoji,
rarity: rarity,
price: Math.max(1, targetPrice),
tier: rarity === "gold" ? "ultra" : (rarity === "red" || rarity === "pink" ? "profit" : "cheap")
};
}

function removeInventoryItemSafely(idx) {
if (upgSelectedInvIdx === idx) upgSelectedInvIdx = null;
else if (upgSelectedInvIdx !== null && idx < upgSelectedInvIdx) upgSelectedInvIdx--;
if (magicSelectedInvIdx === idx) magicSelectedInvIdx = null;
else if (magicSelectedInvIdx !== null && idx < magicSelectedInvIdx) magicSelectedInvIdx--;
contractIndices = contractIndices
.filter(i => i !== idx)
.map(i => (i > idx ? i - 1 : i));
inventory.splice(idx, 1);
}
